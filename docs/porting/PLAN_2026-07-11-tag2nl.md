```csharp
[RelayCommand]
private async Task RunTag2NlAsync()
{
    if (IsBusy || IsBatchRunning) return;
    if (_dataset is null || string.IsNullOrWhiteSpace(_dataset.DatasetRoot))
    {
        StatusText = "请先打开数据集文件夹。";
        return;
    }

    var window = GetMainWindow();
    if (window is null) return;

    // Unsaved: if project has dirty detection, prompt and SaveAll; else skip.
    // Match Form1: save all then continue.

    if (!HasValidTag2NlSettings(_settings.Llm))
    {
        StatusText = "请先在设置中配置有效的 LLM Endpoint 与 Vision 模型。";
        await OpenSettingsAsync(); // existing command method if accessible
        return;
    }

    IsBusy = true;
    CaptionScanResult scan;
    try
    {
        StatusText = "TAG2NL：扫描中…";
        scan = await CaptionGenerationService.ScanDirectoryAsync(_dataset.DatasetRoot);
    }
    catch (Exception ex)
    {
        StatusText = "TAG2NL 扫描失败: " + ex.Message;
        IsBusy = false;
        return;
    }

    IsBusy = false; // allow dialog interaction

    var confirmVm = new Tag2NlConfirmViewModel(scan);
    var confirm = new Views.Tag2NlConfirmWindow { DataContext = confirmVm };
    var ok = await confirm.ShowDialog<bool?>(window);
    if (ok != true) { StatusText = "已取消 TAG2NL。"; return; }

    bool skipExisting = !confirmVm.ReprocessExisting;
    if (skipExisting && scan.Pending == 0)
    {
        StatusText = $"TAG2NL 完成 · 成功 0 · 跳过 {scan.Existing} · 失败 0";
        return;
    }

    var progressVm = new Tag2NlProgressViewModel();
    var progressWin = new Views.Tag2NlProgressWindow { DataContext = progressVm };
    var progress = new Progress<CaptionGenerationProgress>(p =>
    {
        // Ensure UI thread
        global::Avalonia.Threading.Dispatcher.UIThread.Post(() => progressVm.ApplyProgress(p));
    });

    CaptionGenerationResult result;
    IsBusy = true;
    var showTask = progressWin.ShowDialog(window);

    try
    {
        using var client = new OpenAiVisionClient(_settings.Llm);
        var service = new CaptionGenerationService(async (req, ct) =>
        {
            var completion = await client.CompleteAsync(new OpenAiVisionCompletionRequest
            {
                SystemPrompt = req.SystemPrompt,
                UserPrompt = req.UserPrompt,
                ImageData = req.ImageData,
                ContentType = req.ContentType,
            }, ct);
            if (!completion.Success)
                return new CaptionModelResponse(string.Empty, completion.ErrorMessage ?? "LLM error");
            return new CaptionModelResponse(completion.Text, string.Empty);
        });

        result = await service.ProcessAsync(
            scan,
            new CaptionGenerationOptions
            {
                SystemPrompt = string.IsNullOrWhiteSpace(_settings.Llm.Tag2NlSystemPrompt)
                    ? LlmDefaults.Tag2NlSystemPrompt
                    : _settings.Llm.Tag2NlSystemPrompt,
                SkipExisting = skipExisting,
                MaxConcurrency = _settings.Llm.Tag2NlConcurrency,
            },
            progress,
            progressVm.Token);
    }
    catch (Exception ex)
    {
        result = new CaptionGenerationResult { Failed = 1, Errors = { ex.Message } };
    }
    finally
    {
        progressVm.AllowClose = true;
        progressWin.Close();
        IsBusy = false;
    }

    await showTask; // if needed

    string prefix = result.Canceled ? "TAG2NL 已取消" : "TAG2NL 完成";
    StatusText = $"{prefix} · 成功 {result.Succeeded} · 跳过 {result.Skipped} · 失败 {result.Failed} · 输出 {scan.OutputRoot}";
    // If errors, optionally show a simple Window with first 5 lines — keep minimal:
    if (result.Errors.Count > 0)
        StatusText += " | 错误: " + string.Join("; ", result.Errors.Take(5));
}
```

**Fix dialog lifecycle carefully:** Avalonia `ShowDialog` awaits until close. Prefer:

1. Start `ProcessAsync` **after** showing progress window via `Opened` event, **or**
2. `Show()` non-modal progress + await process + `Close()` — simpler and OK for Linux MVP.

**Recommended simpler pattern (use this if ShowDialog races):**

```csharp
progressWin.Show(window);
try {
  result = await service.ProcessAsync(..., progressVm.Token);
} finally {
  progressVm.AllowClose = true;
  progressWin.Close();
}
```
